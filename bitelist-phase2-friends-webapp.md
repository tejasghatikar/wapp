# BiteList — Phase 2: Friends + Webapp

> Build spec for Claude Code. Builds on the working bot from Phase 1.
> Two additions: (1) a friend graph with social bot commands, (2) two webapp pages — your list and a group-decision compare view.
> Webapp is read-only. Bot remains the only way to save or manage your list.

---

## What You're Building

```
Phase 1 (done): Save → Query (solo)
Phase 2 (this):  Share → Connect → Suggest together (social)
```

**Two webapp pages:**
- `/list/:slug` — your saved places + notes. Public. Has a "Connect with me" button.
- `/compare/:slug1/:slug2` — overlap of two people's lists. Generated by bot. Has "Share to WhatsApp" button.

**Four new bot commands:**
- `connect with <name> <slug>` — triggered by the "Connect" button on the list page
- `accept <name>` / `decline <name>` — friendship confirmation
- `suggest with <name>` — shows places both friends saved
- `discover with <name>` — shows what your friend saved that you haven't

## 1. Database Changes

Add to Supabase SQL editor:

```sql
-- Pending connection requests (one-way until accepted)
create table friend_requests (
  id uuid primary key default uuid_generate_v4(),
  requester_id uuid references users(id) on delete cascade,
  recipient_id uuid references users(id) on delete cascade,
  status text default 'pending',   -- 'pending' | 'accepted' | 'declined'
  created_at timestamptz default now(),
  unique(requester_id, recipient_id)
);

-- Confirmed friendships (always bidirectional — insert both directions)
create table friendships (
  id uuid primary key default uuid_generate_v4(),
  user_a_id uuid references users(id) on delete cascade,
  user_b_id uuid references users(id) on delete cascade,
  created_at timestamptz default now(),
  unique(user_a_id, user_b_id)
);

create index idx_friendships_a on friendships(user_a_id);
create index idx_friendships_b on friendships(user_b_id);

-- Compare sessions: bot creates these, returns a URL to user
create table compare_sessions (
  id uuid primary key default uuid_generate_v4(),
  slug_a text not null references users(share_slug),
  slug_b text not null references users(share_slug),
  created_at timestamptz default now()
);
```

## 2. New DB Repository Methods

Add to `src/services/db.js`:

```javascript
// ── Friends ──────────────────────────────────────────────────────────────

export async function createFriendRequest(requesterId, recipientId) {
  const { data, error } = await supabase
    .from('friend_requests')
    .insert({ requester_id: requesterId, recipient_id: recipientId })
    .select()
    .single();
  if (error?.code === '23505') return { duplicate: true };
  if (error) throw error;
  return data;
}

export async function getPendingRequest(requesterId, recipientId) {
  const { data } = await supabase
    .from('friend_requests')
    .select('*')
    .eq('requester_id', requesterId)
    .eq('recipient_id', recipientId)
    .eq('status', 'pending')
    .maybeSingle();
  return data;
}

export async function getPendingRequestByName(recipientId, requesterName) {
  // Find pending request where requester's display_name ilike requesterName
  const { data } = await supabase
    .from('friend_requests')
    .select('*, requester:requester_id(id, whatsapp_number, display_name, share_slug)')
    .eq('recipient_id', recipientId)
    .eq('status', 'pending')
    .ilike('requester.display_name', `%${requesterName}%`)
    .maybeSingle();
  return data;
}

export async function acceptFriendRequest(requestId, requesterId, recipientId) {
  // Update request status
  await supabase
    .from('friend_requests')
    .update({ status: 'accepted' })
    .eq('id', requestId);

  // Insert bidirectional friendship
  await supabase.from('friendships').insert([
    { user_a_id: requesterId, user_b_id: recipientId },
    { user_a_id: recipientId, user_b_id: requesterId }
  ]);
}

export async function declineFriendRequest(requestId) {
  await supabase
    .from('friend_requests')
    .update({ status: 'declined' })
    .eq('id', requestId);
}

export async function getFriends(userId) {
  const { data } = await supabase
    .from('friendships')
    .select('*, friend:user_b_id(id, display_name, share_slug, whatsapp_number)')
    .eq('user_a_id', userId);
  return (data || []).map(row => row.friend);
}

export async function areFriends(userAId, userBId) {
  const { data } = await supabase
    .from('friendships')
    .select('id')
    .eq('user_a_id', userAId)
    .eq('user_b_id', userBId)
    .maybeSingle();
  return !!data;
}

// ── Mutual saves ─────────────────────────────────────────────────────────

export async function getMutualSaves(userAId, userBId) {
  // Places both have saved, ordered by user A's rating
  const { data } = await supabase.rpc('mutual_saves', {
    user_a: userAId,
    user_b: userBId
  });
  return data || [];
}

export async function getNewForUser(userId, friendId) {
  // Places friend saved that userId hasn't
  const { data } = await supabase.rpc('new_for_user', {
    me: userId,
    friend: friendId
  });
  return data || [];
}

// ── Compare sessions ─────────────────────────────────────────────────────

export async function createCompareSession(slugA, slugB) {
  const { data, error } = await supabase
    .from('compare_sessions')
    .insert({ slug_a: slugA, slug_b: slugB })
    .select()
    .single();
  if (error) throw error;
  return data;
}

export async function getUserBySlug(slug) {
  const { data } = await supabase
    .from('users')
    .select('*')
    .eq('share_slug', slug)
    .maybeSingle();
  return data;
}
```

## 3. Supabase SQL Functions (Mutual Saves)

Paste into Supabase SQL editor:

```sql
-- Places both users saved
create or replace function mutual_saves(user_a uuid, user_b uuid)
returns table (
  id uuid, restaurant_name text, area text, google_rating numeric,
  price_level int, cuisine_tags text[], google_maps_url text
)
language sql stable as $$
  select s1.id, s1.restaurant_name, s1.area, s1.google_rating,
         s1.price_level, s1.cuisine_tags, s1.google_maps_url
  from saves s1
  join saves s2 on s2.google_place_id = s1.google_place_id
  where s1.user_id = user_a
    and s2.user_id = user_b
    and s1.deleted_at is null
    and s2.deleted_at is null
  order by coalesce(s1.google_rating, 0) desc
  limit 10;
$$;

-- Places friend has saved that 'me' hasn't
create or replace function new_for_user(me uuid, friend uuid)
returns table (
  id uuid, restaurant_name text, area text, google_rating numeric,
  price_level int, cuisine_tags text[], google_maps_url text
)
language sql stable as $$
  select s.id, s.restaurant_name, s.area, s.google_rating,
         s.price_level, s.cuisine_tags, s.google_maps_url
  from saves s
  where s.user_id = friend
    and s.deleted_at is null
    and not exists (
      select 1 from saves s2
      where s2.user_id = me
        and s2.google_place_id = s.google_place_id
        and s2.deleted_at is null
    )
  order by coalesce(s.google_rating, 0) desc
  limit 10;
$$;
```

## 4. New Bot Commands

### 4.1 Command: `connect with <name> <slug>`

This is triggered by the "Connect with me" button on the list page (via wa.me deep link — see §6.1). The visitor's WhatsApp opens with this pre-filled text and sends it to the bot.

Add to `src/handlers/commands.js`:

```javascript
// Triggered by: "connect with Tejas abc123def"
// Format: "connect with <display_name> <share_slug>"
export async function handleConnectRequest(requester, text) {
  // Parse: "connect with Tejas abc1234"
  const match = text.match(/^connect with (.+?)\s+([a-f0-9]{10,})$/i);
  if (!match) {
    await sendMessage(requester.whatsapp_number,
      "Connection format not recognised. Use the Connect button on someone's BiteList page.");
    return;
  }

  const [, ownerName, slug] = match;
  const owner = await getUserBySlug(slug);

  if (!owner) {
    await sendMessage(requester.whatsapp_number, "Couldn't find that BiteList. The link may have expired.");
    return;
  }

  if (owner.id === requester.id) {
    await sendMessage(requester.whatsapp_number, "That's your own list. 😄");
    return;
  }

  const already = await areFriends(requester.id, owner.id);
  if (already) {
    await sendMessage(requester.whatsapp_number,
      `You and ${owner.display_name || 'that person'} are already connected.`);
    return;
  }

  const req = await createFriendRequest(requester.id, owner.id);
  if (req.duplicate) {
    await sendMessage(requester.whatsapp_number,
      "You've already sent a request. Waiting for them to accept.");
    return;
  }

  // Notify the owner
  const requesterLabel = requester.display_name || requester.whatsapp_number;
  await sendMessage(owner.whatsapp_number,
    `👋 *${requesterLabel}* wants to connect on BiteList so they can see your saved places and find overlap with theirs.\n\nReply *accept ${requesterLabel.split(' ')[0].toLowerCase()}* or *decline ${requesterLabel.split(' ')[0].toLowerCase()}*.`);

  await sendMessage(requester.whatsapp_number,
    `Request sent to ${owner.display_name || ownerName}. I'll let you know when they accept.`);

  await logEvent(requester.id, 'connect_request_sent', { to: owner.id });
}

// Triggered by: "accept priya" or "decline priya"
export async function handleFriendResponse(user, text) {
  const isAccept = /^accept /i.test(text);
  const name = text.replace(/^(accept|decline)\s+/i, '').trim();

  const pending = await getPendingRequestByName(user.id, name);
  if (!pending) {
    await sendMessage(user.whatsapp_number,
      `No pending request from "${name}". Check the name and try again.`);
    return;
  }

  const requester = pending.requester;

  if (isAccept) {
    await acceptFriendRequest(pending.id, requester.id, user.id);
    await sendMessage(user.whatsapp_number,
      `✅ Connected with *${requester.display_name || name}*!\n\nNow try: *suggest with ${requester.display_name?.split(' ')[0] || name}* to find places you both saved.`);
    await sendMessage(requester.whatsapp_number,
      `✅ *${user.display_name || 'They'}* accepted your BiteList connection!\n\nTry: *suggest with ${user.display_name?.split(' ')[0] || 'them'}*`);
    await logEvent(user.id, 'friend_accepted', { friend_id: requester.id });
  } else {
    await declineFriendRequest(pending.id);
    await sendMessage(user.whatsapp_number, `Declined. They won't be notified.`);
  }
}

// "friends" — list confirmed connections
export async function handleFriendsList(user) {
  const friends = await getFriends(user.id);
  if (!friends.length) {
    await sendMessage(user.whatsapp_number,
      `No connections yet. Share your list and ask friends to tap Connect:\n\n${config.publicUrl}/list/${user.share_slug}`);
    return;
  }
  const lines = friends.map((f, i) =>
    `${i + 1}. ${f.display_name || f.whatsapp_number} → ${config.publicUrl}/list/${f.share_slug}`
  );
  await sendMessage(user.whatsapp_number, `Your connections:\n\n${lines.join('\n')}`);
}
```

### 4.2 Command: `suggest with <name>`

```javascript
// "suggest with rahul" → places both saved → /compare/:slug1/:slug2 URL
export async function handleSuggest(user, text) {
  const name = text.replace(/^suggest with\s+/i, '').trim();
  const friends = await getFriends(user.id);
  const friend = friends.find(f =>
    (f.display_name || '').toLowerCase().includes(name.toLowerCase())
  );

  if (!friend) {
    await sendMessage(user.whatsapp_number,
      `No connection named "${name}". Try *friends* to see your list.`);
    return;
  }

  const mutual = await getMutualSaves(user.id, friend.id);

  if (!mutual.length) {
    await sendMessage(user.whatsapp_number,
      `You and ${friend.display_name} haven't saved any of the same places yet.\n\nTry *discover with ${name}* to see what they've saved that you haven't.`);
    return;
  }

  // Create compare session, return URL
  const session = await createCompareSession(user.share_slug, friend.share_slug);
  const url = `${config.publicUrl}/compare/${user.share_slug}/${friend.share_slug}`;

  const preview = mutual.slice(0, 3).map((s, i) =>
    `${i + 1}. *${s.restaurant_name}*${s.area ? `, ${s.area}` : ''} ${s.google_rating ? `⭐${s.google_rating}` : ''}`
  ).join('\n');

  await sendMessage(user.whatsapp_number,
    `You and ${friend.display_name} both saved *${mutual.length} place${mutual.length > 1 ? 's'  : ''}*:\n\n${preview}${mutual.length > 3 ? `\n...and ${mutual.length - 3} more` : ''}\n\n📋 Full list (shareable):\n${url}`);

  await logEvent(user.id, 'suggest', { friend_id: friend.id, mutual_count: mutual.length });
}

// "discover with rahul" → places friend saved that user hasn't
export async function handleDiscover(user, text) {
  const name = text.replace(/^discover with\s+/i, '').trim();
  const friends = await getFriends(user.id);
  const friend = friends.find(f =>
    (f.display_name || '').toLowerCase().includes(name.toLowerCase())
  );

  if (!friend) {
    await sendMessage(user.whatsapp_number,
      `No connection named "${name}". Try *friends* to see your list.`);
    return;
  }

  const newPlaces = await getNewForUser(user.id, friend.id);

  if (!newPlaces.length) {
    await sendMessage(user.whatsapp_number,
      `You've already saved everything ${friend.display_name} has. They need to save more places!`);
    return;
  }

  const { formatSaveList } = await import('../utils/format.js');
  const header = `${friend.display_name} saved these — you haven't been yet:`;
  await sendMessage(user.whatsapp_number, formatSaveList(newPlaces, header));

  await logEvent(user.id, 'discover', { friend_id: friend.id, new_count: newPlaces.length });
}
```

## 7. Social Augmentation in Regular Queries

When a user runs a query like `places in Indiranagar`, if any result was also saved by a friend, show it. Add to `src/handlers/query.js`:

```javascript
// After getting matched saves from Claude, augment with friend data
async function augmentWithFriendData(userId, saves) {
  const friends = await getFriends(userId);
  if (!friends.length) return saves;

  // Get all friend saves in one query (limit scope for performance)
  const placeIds = saves.map(s => s.google_place_id).filter(Boolean);
  if (!placeIds.length) return saves;

  const friendSaveMap = {}; // place_id → [friend names]

  for (const friend of friends) {
    const { data } = await supabase
      .from('saves')
      .select('google_place_id')
      .eq('user_id', friend.id)
      .in('google_place_id', placeIds)
      .is('deleted_at', null);

    (data || []).forEach(s => {
      if (!friendSaveMap[s.google_place_id]) friendSaveMap[s.google_place_id] = [];
      friendSaveMap[s.google_place_id].push(friend.display_name || 'a friend');
    });
  }

  return saves.map(s => ({
    ...s,
    _friends_who_saved: friendSaveMap[s.google_place_id] || []
  }));
}
```

Update `formatSaveList` in `src/utils/format.js` to render the friend tag:

```javascript
// In the per-save formatting block, after cuisine/price:
if (save._friends_who_saved?.length) {
  const names = save._friends_who_saved.join(', ');
  lines.push(`   👥 Also saved by ${names}`);
}
```
